# Capture is per-OS, so its dependencies are too. The markers keep a macOS
# install from dragging in X11 bindings it can never use, and a Linux install
# from trying to build Objective-C bridges that do not apply.
#
# python-xlib is pure Python and would install harmlessly anywhere, but the
# marker states the intent: it belongs to the x11 backend.
python-xlib==0.33 ; sys_platform == "linux"

# Windows Graphics Capture, the only path measured to return real pixels from
# the game's Direct3D window - GDI's BitBlt hands back pure black, and
# PrintWindow works but costs 20ms a grab against this one's 2us region crop.
# See tools/windows_probe.py, which is what measured that.
windows-capture==2.0.1 ; sys_platform == "win32"
# Window enumeration and geometry. WGC captures the pixels but has nothing to
# say about which window is the game or where it sits on the desktop.
pywin32==312 ; sys_platform == "win32"
